# Fragmentation Governor CUDA proof (optional; requires a CUDA-capable device).
if(FG_BUILD_CUDA)
  find_package(CUDAToolkit QUIET)
  if(CUDAToolkit_FOUND)
    enable_language(CUDA)
    set(CMAKE_CUDA_ARCHITECTURES 120)
    add_executable(fg_cuda_proof fg_cuda_proof.cu)
    target_link_libraries(fg_cuda_proof PRIVATE FragmentationGovernor CUDA::cudart)
    if(MSVC)
      target_compile_options(fg_cuda_proof PRIVATE /W4 /WX)
    endif()
    add_test(NAME fg_cuda_proof COMMAND fg_cuda_proof)
    message(STATUS "FG CUDA proof enabled (device: sm_120)")
  else()
    message(STATUS "FG CUDA proof disabled: CUDA toolkit not found")
  endif()
endif()
